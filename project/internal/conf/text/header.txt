# sky.conf — пути проекта.
# scope — папки проекта от корня репозитория; sky применяет патчи, снимает «до» и коммитит только в них.
#   бинарь, одна папка:  scope = [project/]
#   веб:                 scope = [backend/, frontend/]
# inbox — папка, куда приходят patch*.zip.
# regulation — необязательная ссылка (URL) на регламент версий; sky её только хранит и не загружает.
# build — точка сборки: папка от корня репозитория и команда, которая в ней запускается; строк может быть несколько (бэкенд, фронтенд), sky build гонит их по порядку и снимает сборку (ветку build/ММДД-ЧЧММ), только если все прошли.
#   build = backend/ go build ./...
#   build = frontend/ bun run build
#   проект в корне:     build = ./ make
